# 线程
1️⃣ PCB（进程控制块）：线程的"身份证"

  struct task_struct {                    // 线程的完整信息
     uint32_t* self_kstack;	            // 栈顶位置 ⭐
     enum task_status status;             // 状态：运行/就绪/阻塞
     uint8_t priority;		            // 优先级（固定值）
     char name[16];                       // 线程名字

     uint8_t ticks;	                    // 剩余时间片 ⭐
     uint32_t elapsed_ticks;              // 总运行时间
     struct list_elem general_tag;		// 在就绪队列中的节点 ⭐
     struct list_elem all_list_tag;       // 在全局队列中的节点
     uint32_t* pgdir;                     // 页表地址
     uint32_t stack_magic;	            // 栈溢出检测
  };

  类比：这就是线程的"档案袋"
  - name: 姓名
  - priority: 职级（31 = 高层，8 = 普通员工）
  - ticks: 工作时长配额
  - self_kstack: 工作台位置
  - status: 当前状态（在岗/待命/休息）

  2️⃣ 线程栈（thread_stack）：线程的"断点记忆"

  struct thread_stack {
     uint32_t ebp;                        // 基址寄存器
     uint32_t ebx;                        // 通用寄存器
     uint32_t edi;
     uint32_t esi;

     void (*eip)(thread_func* func, void* func_arg);  // 下次执行的地址 ⭐

     void (*unused_retaddr);              // 占位返回地址
     thread_func* function;               // 线程函数地址 ⭐
     void* func_arg;                      // 线程函数参数 ⭐
  };

  类比：演员的"剧本书签"
  - eip: 标记演到第几页
  - ebp/ebx/edi/esi: 记录道具位置（寄存器状态）
  - function: 剧本名称
  - func_arg: 台词内容

  3️⃣ 中断栈（intr_stack）：中断时的"现场快照"

  struct intr_stack {
      uint32_t vec_no;	               // 中断号
      uint32_t edi;                       // 保存所有寄存器 ⭐
      uint32_t esi;
      uint32_t ebp;
      uint32_t esp_dummy;
      uint32_t ebx;
      uint32_t edx;
      uint32_t ecx;
      uint32_t eax;
      uint32_t gs;
      uint32_t fs;
      uint32_t es;
      uint32_t ds;

      uint32_t err_code;
      void (*eip)(void);                  // 中断时的执行位置 ⭐
      uint32_t cs;
      uint32_t eflags;
      void* esp;
      uint32_t ss;
  };

  类比：拍摄现场的"暂停画面"
  - 当导演喊"cut"时，摄像机记录：
    - 演员位置（所有寄存器）
    - 剧本进度（eip）
    - 道具状态（段寄存器）

  🎬 第四步：线程切换的完整流程

  让我用时间轴展示三个线程如何切换：

  时间 →  0ms     10ms    20ms    30ms    40ms    50ms
          │       │       │       │       │       │
  Main    ████████│       │       ████████│       │
                  │       │       │       │       │
  Thread_A        │████████│       │       │████████
                  │       │       │       │       │
  Thread_B        │       │████████│       │       │████████

  每 10ms 发生一次时钟中断，触发调度

  详细过程（以 Main → Thread_A 为例）：

  1. 时钟中断触发（10ms到了）
  ; CPU 自动做：
  push ss              ; 保存栈段
  push esp             ; 保存栈指针
  push eflags          ; 保存标志
  push cs              ; 保存代码段
  push eip             ; 保存当前执行位置 ⭐
  2. 进入中断处理程序
  ; kernel.S 中的中断处理：
  pushad               ; 保存所有通用寄存器
  push ds              ; 保存段寄存器
  push es
  push fs
  push gs
  3. → 形成完整的 intr_stack！
  4. 调用调度器 schedule()
  void schedule(void) {
      struct task_struct* cur = running_thread();   // 当前线程（Main）

      if (cur->status == TASK_RUNNING) {
          // 时间片用完了
          cur->status = TASK_READY;                // 改为就绪
          list_append(&thread_ready_list, &cur->general_tag);  // 放回就绪队列
      }

      struct task_struct* next = list_pop(&thread_ready_list);  // 取下一个（Thread_A）
      next->status = TASK_RUNNING;

      switch_to(cur, next);  // ⭐ 关键：切换栈指针！
  }
  5. switch_to 汇编魔法
  switch_to:
      ; 保存当前线程（Main）的上下文
      push esi
      push edi
      push ebx
      push ebp
      mov [eax], esp     ; 保存 Main 的栈顶到 PCB ⭐
      
      ; 切换到下一个线程（Thread_A）
      mov esp, [edx]     ; 加载 Thread_A 的栈顶 ⭐
      pop ebp
      pop ebx
      pop edi
      pop esi
      ret                ; 跳转到 Thread_A 的 eip！
  6. Thread_A 开始运行
    - CPU 现在执行 console_put_str("argA ")
    - 10ms 后又被中断，重复流程

  🔑 关键问题解答

  Q1: 线程和函数的区别？

  | 特性   | 普通函数     | 线程            |
  |------|----------|---------------|
  | 调用方式 | call 指令  | 调度器 switch_to |
  | 返回方式 | ret 到调用者 | 永不返回（无限循环）    |
  | 栈    | 使用调用者的栈  | 有自己独立的栈       |
  | 上下文  | 共享寄存器    | 独立保存/恢复寄存器    |
  | 生命周期 | 执行完就结束   | 一直运行（直到被杀死）   |

  Q2: 优先级怎么用？

  thread_start("k_thread_a", 31, k_thread_a, "argA ");  // 优先级 31
  thread_start("k_thread_b", 8, k_thread_b, "argB ");   // 优先级 8

  数字越大，优先级越高！

  - 优先级决定时间片大小：
  pthread->ticks = prio;  // 优先级 31 → 每次运行 31 个时钟周期
  - 每次时钟中断：
  current_thread->ticks--;
  if (ticks == 0) {
      schedule();  // 时间片用完，切换
  }

  效果：Thread_A（优先级31）运行更久，Thread_B（优先级8）运行更短。

  Q3: intr_enable() 为什么关键？

  intr_enable();  // 开中断！
  while(1) {
      console_put_str("Main ");
  };

  如果不开中断：
  - 时钟中断无法触发
  - schedule() 永远不会被调用
  - Main 会独占 CPU，其他线程永远运行不了！

  开中断后：
  - 每 10ms 自动调用 schedule()
  - 三个线程轮流运行
